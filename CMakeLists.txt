cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(RegistryGuard VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Visual Studio C++ tools on Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(GUARD_ASAN "Enable AddressSanitizer in user-mode checks" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(GUARD_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(RegistryGuard SDK/RegistryGuard.cpp)
target_link_libraries(RegistryGuard PUBLIC setupapi)
add_executable(RegistryGuardApp WIN32 App/Main.cpp)
target_link_libraries(RegistryGuardApp PRIVATE RegistryGuard user32 gdi32 comctl32 comdlg32 shell32)
add_executable(RegistryGuardCLI App/Console.cpp)
target_link_libraries(RegistryGuardCLI PRIVATE RegistryGuard)
add_executable(RegistryGuardExample Examples/policy.cpp)
target_link_libraries(RegistryGuardExample PRIVATE RegistryGuard)
add_executable(RegistryGuardTests Tests/Protocol.cpp)
target_link_libraries(RegistryGuardTests PRIVATE RegistryGuard)
enable_testing()
add_test(NAME OfflineProtocolAndBounds COMMAND RegistryGuardTests)
set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque Microsoft-signed INF/CAT/SYS package")
set(PACKAGE_RESOURCE "")
if(DRIVER_PACKAGE_DIRECTORY)
 enable_language(RC)
 foreach(name TitanRegistryGuard.inf TitanRegistryGuard.cat TitanRegistryGuard.sys)
  if(NOT EXISTS "${DRIVER_PACKAGE_DIRECTORY}/${name}")
   message(FATAL_ERROR "Incomplete opaque package: ${name}")
  endif()
 endforeach()
 set(PACKAGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/DriverPackage.rc")
 file(WRITE "${PACKAGE_RESOURCE}" "101 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanRegistryGuard.inf\"\n102 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanRegistryGuard.cat\"\n103 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanRegistryGuard.sys\"\n")
endif()
add_library(SetupCore Installer/Setup.cpp)
target_link_libraries(SetupCore PUBLIC setupapi newdev wintrust crypt32 advapi32 ole32)
add_executable(RegistryGuardSetup WIN32 Installer/Main.cpp ${PACKAGE_RESOURCE})
add_executable(RegistryGuardSetupConsole Installer/Console.cpp ${PACKAGE_RESOURCE})
foreach(target RegistryGuardSetup RegistryGuardSetupConsole)
 target_link_libraries(${target} PRIVATE SetupCore user32 gdi32)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()
foreach(target RegistryGuardApp RegistryGuardCLI RegistryGuardExample RegistryGuardTests RegistryGuardSetup RegistryGuardSetupConsole)
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
