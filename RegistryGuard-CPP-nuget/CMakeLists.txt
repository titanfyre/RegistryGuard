cmake_minimum_required(VERSION 3.24)
project(RegistryGuardSDK VERSION 0.1.1 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This SDK requires Windows x64. No driver is installed by this build.")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(TitanRegistryGuard STATIC
  "sdk/SDK/RegistryGuard.cpp"
)
add_library(RegistryGuardSDK::sdk ALIAS TitanRegistryGuard)
set_target_properties(TitanRegistryGuard PROPERTIES EXPORT_NAME sdk)
target_compile_features(TitanRegistryGuard PUBLIC cxx_std_17)
target_compile_definitions(TitanRegistryGuard PUBLIC )
target_compile_definitions(TitanRegistryGuard PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(TitanRegistryGuard PRIVATE /utf-8 /EHsc /W4)
endif()
target_include_directories(TitanRegistryGuard PUBLIC
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/sdk>"
  "$<INSTALL_INTERFACE:include/>"
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/sdk/SDK>"
  "$<INSTALL_INTERFACE:include/SDK>"
)
target_link_libraries(TitanRegistryGuard PUBLIC advapi32 comctl32 comdlg32 crypt32 gdi32 newdev ole32 setupapi shell32 user32 wintrust)
install(TARGETS TitanRegistryGuard EXPORT RegistryGuardSDKTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY sdk/ DESTINATION include FILES_MATCHING PATTERN "*.h" PATTERN "*.hpp" PATTERN "*.inl")
install(FILES LICENSE DESTINATION share/registryguard)
install(EXPORT RegistryGuardSDKTargets NAMESPACE RegistryGuardSDK:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/RegistryGuardSDK)
configure_file("RegistryGuardSDKConfig.cmake.in" "${CMAKE_CURRENT_BINARY_DIR}/RegistryGuardSDKConfig.cmake" @ONLY)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/RegistryGuardSDKConfigVersion.cmake" VERSION 0.1.1 COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/RegistryGuardSDKConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/RegistryGuardSDKConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/RegistryGuardSDK)
