cmake_minimum_required(VERSION 3.24)
project(ProductJsonNode LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(PRODUCT_BRIDGE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../RegistryGuard-C-ffi" CACHE PATH "Product C ABI")
add_subdirectory("${PRODUCT_BRIDGE_SOURCE}" bridge)
add_library(registryguard_json_node MODULE native.cpp DelayLoad.cpp)
set_target_properties(registryguard_json_node PROPERTIES PREFIX "" SUFFIX ".node")
target_compile_features(registryguard_json_node PRIVATE cxx_std_17)
target_compile_definitions(registryguard_json_node PRIVATE NAPI_VERSION=8 NODE_GYP_MODULE_NAME=registryguard_json_node)
target_include_directories(registryguard_json_node PRIVATE "${NODE_HEADERS}")
target_link_libraries(registryguard_json_node PRIVATE RegistryGuardJsonSDK "${NODE_LIBRARY}" delayimp)
target_link_options(registryguard_json_node PRIVATE /DELAYLOAD:node.exe)
